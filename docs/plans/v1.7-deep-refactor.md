# v1.7 深度重构方案

> 目标：把 v1.6 阶段妥协的「双语言 cinterop 胶水代码」换成「真正的跨端共享」。
> 原则：每步独立 commit + build 验证 + 回滚友好。

## 背景

v1.6 阶段 Shared 模块已经有 Models + Services，但 iOS 接入时遇到几个问题：
- `StateFlow<List<X>>` 在 Swift 侧不暴露 `AsyncSequence`，iOS wrapper 只能 100ms 轮询 `value`
- Models 在 Shared 用 `Long?` 表示时间，iOS 原生 Swift struct 用 `Date?`，需要 137 行转换层
- iOS AfdianAPIService 保留 Swift Codable 原生实现，未切到 Shared.Ktor

虽然代码可工作，但不优雅。

## Phase A: Shared.DatabaseService callback API

### A.1 Shared 侧（DatabaseService.kt）

**新 API**：
```kotlin
class DatabaseService {
    interface Listener {
        fun onCreatorsChanged(creators: List<Creator>)
        fun onAlbumsChanged(albums: List<Album>)
        fun onAudioItemsChanged(items: List<AudioItem>)
    }
    
    fun addListener(listener: Listener)
    fun removeListener(listener: Listener)
    
    // CRUD 保持不变
    fun upsertCreator(creator: Creator)
    fun upsertCreators(creators: List<Creator>)
    fun deleteCreator(id: String)
    fun selectedCreators(): List<Creator>
    fun upsertAlbum(album: Album)
    fun upsertAlbums(albums: List<Album>)
    fun albumsForCreator(creatorId: String): List<Album>
    fun upsertAudioItem(item: AudioItem)
    fun upsertAudioItems(items: List<AudioItem>)
    fun audioItemsForAlbum(albumId: String): List<AudioItem>
    fun audioItemById(id: String): AudioItem?
    fun clearAll()
    
    // 内部保留 StateFlow 给 Android 用（不暴露给 Swift）
    internal val creators: StateFlow<List<Creator>>
    internal val albums: StateFlow<List<Album>>
    internal val audioItems: StateFlow<List<AudioItem>>
}
```

**实现**：每次 `_creators.update { ... }` 后遍历 `listeners` 调用 `onCreatorsChanged`。

### A.2 Android 侧（AudioPlayerManager.kt + 各 view）

- AudioPlayerManager 用 `_state` 时通过 `addListener` 接收更新（替代轮询）
- 各 view 用 `db.creators.value` 的地方保持不变（StateFlow 还在）

### A.3 iOS 侧（DatabaseService.swift）

```swift
@Observable
final class DatabaseService {
    static let shared = DatabaseService()
    
    private(set) var creators: [Creator] = []
    private(set) var albums: [Album] = []
    private(set) var audioItems: [AudioItem] = []
    
    private let backend = Shared.DatabaseService.companion.shared
    private let listener = DatabaseListener()  // 内部类实现 Shared.DatabaseService.Listener
    
    init() {
        // 初始化时同步一次当前值
        creators = (backend.creators.value as? [Shared.Creator])?.map { Creator($0) } ?? []
        albums = (backend.albums.value as? [Shared.Album])?.map { Album($0) } ?? []
        audioItems = (backend.audioItems.value as? [Shared.AudioItem])?.map { AudioItem($0) } ?? []
        
        // 注册 callback
        backend.addListener(listener)
    }
    
    deinit {
        backend.removeListener(listener)
    }
    
    // CRUD 转发（API 形态保持兼容）
    func upsertCreator(_ creator: Creator) { backend.upsertCreator(creator: creator.toSharedCreator()) }
    // ... 其他类似
}

private class DatabaseListener: Shared.DatabaseService.Listener {
    func onCreatorsChanged(creators: [Shared.Creator]) {
        // 在主线程更新 @Observable 属性
        Task { @MainActor in
            (DatabaseService.shared as Any).setValue(Creator(...), forKey: "creators")
        }
    }
    // ...
}
```

或者更简洁：让 DatabaseListener 直接持有 DatabaseService.shared 的引用（用 weak）：
```swift
class DatabaseListener: Shared.DatabaseService.Listener {
    weak var service: DatabaseService?
    func onCreatorsChanged(creators: [Shared.Creator]) {
        Task { @MainActor in
            self.service?.creators = creators.map { Creator($0) }
        }
    }
}
```

**预期代码量**：从 65 行 wrapper（含 100ms 轮询 task）降到 ~35 行（含 listener 桥接）。

```swift
// Swift 友好的访问器（Date / URL 包装）

    var lastSyncedAtDate: Date? {
        get { lastSyncedAt?.let { Date(timeIntervalSince1970: TimeInterval($0.int64Value) / 1000.0) } }
        set { lastSyncedAt = newValue.map { KotlinLong(value: Int64($0.timeIntervalSince1970 * 1000.0)) } }
    }
    var afdianPageURL: URL? { URL(string: afdianPageUrl) }
}

extension Album {
    var lastSyncedAtDate: Date? { /* 同上 */ }
}

extension AudioItem {
    var publishTimeDate: Date {
        get { Date(timeIntervalSince1970: TimeInterval(publishTime) / 1000.0) }
        set { publishTime = Int64(newValue.timeIntervalSince1970 * 1000.0) }
    }
    func progressRatio(progress: Double) -> Double {
        if duration <= 0 { return 0 }
        return min(1.0, progress / duration)
    }
}
```

## Phase C: Shared.AfdianApiService + iOS 调用

### C.1 对齐 JSON 模型

当前差异：
- iOS Swift Codable：`snake_case` 字段名（如 `user_id`）
- Shared Kotlinx.serialization：`camelCase` + `@SerialName("user_id")`

**两种方案**：
1. **统一用 camelCase**：服务端 JSON 已经是 snake_case，需要在 iOS 用 CodingKeys 转，或在 Shared 用 `@SerialName`
2. **Shared 用 snake_case**：对齐 iOS 当前行为，但 Kotlin 代码丑

**选方案 1**：Shared 加 `@SerialName("user_id")`，iOS 删除 CodingKeys 转写。这对齐现有 iOS JSON 解析逻辑，最小改动。

### C.3 iOS AfdianAPIService 改用 Shared

```swift
final class AfdianAPIService {
    static let shared = AfdianAPIService()
    
    private let api = Shared.AfdianApiService()  // 传入 SecureStorage
    private let secureStorage = Shared.SecureStorage()
    
    private var authToken: String? {
        secureStorage.get(key: AfdianAPIService.authTokenKey)
    }
    
    func fetchSponsoringCreators() async throws -> [Creator] {
        let sharedList = try await api.fetchSponsoringCreators()
        return sharedList.map { Creator($0) }  // typealias, 直接转换
    }
    
    // 类似其他方法
}
```
